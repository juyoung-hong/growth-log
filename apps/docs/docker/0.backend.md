```zsh
docker build -t growth-log-backend .
```

```zsh
docker run -d --name growth-log-backend -p 8000:8000 \
  -e DB_USER=<실제 값> \
  -e DB_PASSWORD=<실제 값> \
  -e DB_DSN=<실제 값, 예: oracleadb_medium> \
  -e TNS_ADMIN=/app/wallet \
  -e OCI_BUCKET_NAME=<실제 값> \
  -e OCI_AUTONOMOUS_DATABASE_ID=<실제 값> \
  -v <WALLET_DIR>:/app/wallet:ro \
  -v ~/.oci:/app/.oci:ro \
  growth-log-backend

curl -i http://localhost:8000/health
curl -i http://localhost:8000/api/v1/persons
```
