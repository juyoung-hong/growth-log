```zsh
kubectl create secret generic backend-wallet -n growth-log \
  --from-file=<지갑 디렉터리 경로>
```

```zsh
# 원래 위 명령어를 사용해야하지만 k8s Secret의 키 이름 규칙([-._a-zA-Z0-9]+)이 :를 허용하지 않으므로 아래 명령어로 대체함
kubectl create secret generic backend-wallet -n growth-log \
  $(find ~/.oracle/Wallet_ORACLEADB -maxdepth 1 -type f ! -name '*:Zone.Identifier' -printf '--from-file=%p\n')
```
